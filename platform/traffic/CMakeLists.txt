cmake_minimum_required(VERSION 3.24)
project(skewer_traffic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(skewer-packet-tuple packet_tuple.cpp)
add_library(skewer-packet-rewrite packet_rewrite.cpp)
target_link_libraries(skewer-packet-rewrite PUBLIC skewer-packet-tuple)
add_executable(packet_rewrite_test packet_rewrite_test.cpp)
target_link_libraries(packet_rewrite_test PRIVATE skewer-packet-rewrite)
if(NOT MSVC)
    target_compile_options(skewer-packet-rewrite PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(packet_rewrite_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MINGW)
    target_link_options(packet_rewrite_test PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_library(skewer-flow-index flow_index.cpp)
target_link_libraries(skewer-flow-index PUBLIC skewer-packet-tuple)
add_executable(flow_index_test flow_index_test.cpp)
target_link_libraries(flow_index_test PRIVATE skewer-flow-index)
target_include_directories(skewer-packet-tuple PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(packet_tuple_test packet_tuple_test.cpp)
target_link_libraries(packet_tuple_test PRIVATE skewer-packet-tuple)
if(NOT MSVC)
    target_compile_options(skewer-flow-index PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(flow_index_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(skewer-packet-tuple PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(packet_tuple_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MINGW)
    target_link_options(flow_index_test PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(packet_tuple_test PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_test(NAME packet_rewrite COMMAND packet_rewrite_test)
