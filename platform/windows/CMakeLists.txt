cmake_minimum_required(VERSION 3.24)
project(skewer_windows_backend LANGUAGES CXX)

# Usable standalone for backend tests, or selected by the root application build.
if(NOT WIN32)
    message(FATAL_ERROR "Use a Windows compiler or mingw-toolchain.cmake")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(skewer-windows-processes STATIC processes.cpp)
target_include_directories(skewer-windows-processes PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(skewer-windows-processes PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(skewer-windows-processes PUBLIC advapi32 user32 shell32)
if(MSVC)
    target_compile_options(skewer-windows-processes PRIVATE /W4 /permissive-)
else()
    target_compile_options(skewer-windows-processes PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(skewer-process-probe probe.cpp)
target_link_libraries(skewer-process-probe PRIVATE skewer-windows-processes)
if(MINGW)
    # A self-contained diagnostic binary, requiring no MinGW DLL installation.
    target_link_options(skewer-process-probe PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_library(skewer-app-egress-guard STATIC app_egress_guard.cpp)
target_include_directories(skewer-app-egress-guard PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(skewer-app-egress-guard PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(skewer-app-egress-guard PUBLIC fwpuclnt rpcrt4)
add_executable(app_egress_guard_test app_egress_guard_test.cpp)
target_link_libraries(app_egress_guard_test PRIVATE skewer-app-egress-guard ws2_32)
if(MINGW)
    target_compile_options(skewer-app-egress-guard PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(app_egress_guard_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(app_egress_guard_test PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
# Explicit elevated VM-only fixture; persistent policy changes never run in CTest.
add_test(NAME windows_app_guard_inputs COMMAND app_egress_guard_test --self-test)
set_tests_properties(windows_app_guard_inputs PROPERTIES TIMEOUT 10)

add_library(skewer-windows-udp STATIC udp_endpoint.cpp)
target_include_directories(skewer-windows-udp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(skewer-windows-udp PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(skewer-windows-udp PUBLIC ws2_32)
if(MINGW)
    target_compile_options(skewer-windows-udp PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(udp_endpoint_test udp_endpoint_test.cpp)
target_link_libraries(udp_endpoint_test PRIVATE skewer-windows-udp)
if(MINGW)
    target_link_options(udp_endpoint_test PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME windows_udp COMMAND udp_endpoint_test)
set_tests_properties(windows_udp PROPERTIES TIMEOUT 30)

# Consume the real protocol sources, not a copied Windows implementation.
set(TUNTOM_SOURCE_DIR "" CACHE PATH "Tuntom checkout with portable session headers")
if(TUNTOM_SOURCE_DIR)
    if(NOT EXISTS "${TUNTOM_SOURCE_DIR}/src/packet_types.hpp")
        message(FATAL_ERROR "TUNTOM_SOURCE_DIR requires the portable Tuntom session headers")
    endif()
    add_library(skewer-tuntom-session INTERFACE)
    target_include_directories(skewer-tuntom-session SYSTEM INTERFACE "${TUNTOM_SOURCE_DIR}/src")
    target_compile_definitions(skewer-tuntom-session INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(skewer-tuntom-session INTERFACE bcrypt)
    add_library(skewer-tunnel-pump STATIC tunnel_pump.cpp)
    target_link_libraries(skewer-tunnel-pump PUBLIC skewer-windows-udp skewer-tuntom-session)
    if(MINGW)
        target_compile_options(skewer-tunnel-pump PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_executable(tunnel_pump_test tunnel_pump_test.cpp)
    target_link_libraries(tunnel_pump_test PRIVATE skewer-tunnel-pump)
    if(MINGW)
        target_link_options(tunnel_pump_test PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME windows_tunnel_pump COMMAND tunnel_pump_test)
    set_tests_properties(windows_tunnel_pump PROPERTIES TIMEOUT 30)
    add_executable(tuntom_udp_test tuntom_udp_test.cpp)
    target_link_libraries(tuntom_udp_test PRIVATE skewer-windows-udp skewer-tuntom-session)
    if(MINGW)
        target_compile_options(tuntom_udp_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_link_options(tuntom_udp_test PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME windows_tuntom_udp COMMAND tuntom_udp_test)
    set_tests_properties(windows_tuntom_udp PROPERTIES TIMEOUT 30)
endif()
add_test(NAME windows_processes COMMAND skewer-process-probe --self-test)
set_tests_properties(windows_processes PROPERTIES TIMEOUT 30)

# Optional until the transport supervisor is ported. No network setup runs at build time.
option(SKEWER_WINDOWS_TUN "Build the Wintun packet adapter and tests" OFF)
if(SKEWER_WINDOWS_TUN)
    find_path(WINTUN_INCLUDE_DIR wintun.h REQUIRED)
    add_library(skewer-wintun STATIC wintun_device.cpp)
    target_include_directories(skewer-wintun SYSTEM PUBLIC ${WINTUN_INCLUDE_DIR})
    target_include_directories(skewer-wintun PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(skewer-wintun PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MINGW)
        target_compile_options(skewer-wintun PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_executable(wintun_device_test wintun_device_test.cpp)
    target_link_libraries(wintun_device_test PRIVATE skewer-wintun)
    if(MINGW)
        target_link_options(wintun_device_test PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME wintun_device COMMAND wintun_device_test)
    set_tests_properties(wintun_device PROPERTIES TIMEOUT 30)
    if(TARGET skewer-tunnel-pump)
        # Explicit elevated VM-only probe, intentionally excluded from CTest.
        add_executable(tunnel_pump_live tunnel_pump_live.cpp)
        target_link_libraries(tunnel_pump_live PRIVATE skewer-tunnel-pump skewer-wintun)
        if(MINGW)
            target_compile_options(tunnel_pump_live PRIVATE -Wall -Wextra -Wpedantic -Werror)
            target_link_options(tunnel_pump_live PRIVATE -municode -static -static-libgcc -static-libstdc++)
        endif()
    endif()
endif()

option(SKEWER_WINDOWS_APP "Build the Windows application" OFF)
if(SKEWER_WINDOWS_APP)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Concurrent Network Svg)
    add_subdirectory(../common portable-ui)
    if(NOT DEFINED SKEWER_APP_VERSION)
        set(SKEWER_APP_VERSION "development")
    endif()
    add_executable(skewer WIN32 application.cpp ../../data/icons.qrc)
    set_target_properties(skewer PROPERTIES AUTORCC ON)
    target_include_directories(skewer PRIVATE ../../src)
    target_compile_definitions(skewer PRIVATE SKEWER_VERSION="${SKEWER_APP_VERSION}")
    target_link_libraries(skewer PRIVATE skewer-windows-processes skewer-process-browser Qt6::Widgets Qt6::Concurrent Qt6::Network Qt6::Svg)
    if(MINGW)
        target_compile_options(skewer PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
        # Qt's entry point is built with stack protection enabled.
        target_link_options(skewer PRIVATE -fstack-protector-strong)
    endif()
    install(TARGETS skewer RUNTIME DESTINATION bin)
endif()
