cmake_minimum_required(VERSION 3.24)
project(skewer_windows_backend LANGUAGES CXX)

# Usable standalone for backend tests, or selected by the root application build.
if(NOT WIN32)
    message(FATAL_ERROR "Use a Windows compiler or mingw-toolchain.cmake")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(skewer-windows-processes STATIC processes.cpp)
target_include_directories(skewer-windows-processes PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(skewer-windows-processes PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(skewer-windows-processes PUBLIC advapi32 user32 shell32)
if(MSVC)
    target_compile_options(skewer-windows-processes PRIVATE /W4 /permissive-)
else()
    target_compile_options(skewer-windows-processes PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(skewer-process-probe probe.cpp)
target_link_libraries(skewer-process-probe PRIVATE skewer-windows-processes)
if(MINGW)
    # A self-contained diagnostic binary, requiring no MinGW DLL installation.
    target_link_options(skewer-process-probe PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_test(NAME windows_processes COMMAND skewer-process-probe --self-test)
set_tests_properties(windows_processes PROPERTIES TIMEOUT 30)

option(SKEWER_WINDOWS_APP "Build the Windows application" OFF)
if(SKEWER_WINDOWS_APP)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Concurrent Network Svg)
    add_subdirectory(../common portable-ui)
    if(NOT DEFINED SKEWER_APP_VERSION)
        set(SKEWER_APP_VERSION "development")
    endif()
    add_executable(skewer WIN32 application.cpp ../../data/icons.qrc)
    set_target_properties(skewer PROPERTIES AUTORCC ON)
    target_include_directories(skewer PRIVATE ../../src)
    target_compile_definitions(skewer PRIVATE SKEWER_VERSION="${SKEWER_APP_VERSION}")
    target_link_libraries(skewer PRIVATE skewer-windows-processes skewer-process-browser Qt6::Widgets Qt6::Concurrent Qt6::Network Qt6::Svg)
    if(MINGW)
        target_compile_options(skewer PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
        # Qt's entry point is built with stack protection enabled.
        target_link_options(skewer PRIVATE -fstack-protector-strong)
    endif()
    install(TARGETS skewer RUNTIME DESTINATION bin)
endif()
