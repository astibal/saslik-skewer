cmake_minimum_required(VERSION 3.24)
project(skewer_portable_ui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
add_library(skewer-process-browser process_browser.cpp launch_catalog.cpp)
target_link_libraries(skewer-process-browser PUBLIC Qt6::Widgets)
target_include_directories(skewer-process-browser PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(NOT MSVC)
    target_compile_options(skewer-process-browser PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Network)
    add_executable(tunnel_contract_test tunnel_contract_test.cpp)
    target_link_libraries(tunnel_contract_test PRIVATE Qt6::Core Qt6::Network)
    add_test(NAME tunnel_contract COMMAND tunnel_contract_test)
    add_executable(process_browser_test process_browser_test.cpp)
    target_link_libraries(process_browser_test PRIVATE skewer-process-browser Qt6::Test)
    add_test(NAME portable_process_browser COMMAND process_browser_test)
    set_tests_properties(portable_process_browser PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(launch_catalog_test launch_catalog_test.cpp)
    target_link_libraries(launch_catalog_test PRIVATE skewer-process-browser Qt6::Test)
    add_test(NAME launch_catalog COMMAND launch_catalog_test)
    set_tests_properties(launch_catalog PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(application_shell_test application_shell_test.cpp)
    target_link_libraries(application_shell_test PRIVATE Qt6::Widgets)
    add_test(NAME shared_application_shell COMMAND application_shell_test)
    set_tests_properties(shared_application_shell PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
